funcion assert(nombre, condicion) {
    arr_set(runner_stats, 1, runner_stats[1] + 1);
    si (condicion) {
        mostrar("  [PASS] " + nombre);
        arr_set(runner_stats, 0, runner_stats[0] + 1);
        retornar true;
    } sino {
        mostrar("  [FAIL] " + nombre);
        retornar false;
    }
}

assert("Operadores logicos (&&, ||, no)", (true && true) && (false || true) && (no false));

nums = [10, 20, 30, 40, 50];
assert("Acceso por indice [] en listas", nums[0] == 10 && nums[2] == 30 && nums[4] == 50);

persona = {nombre: "PELE", version: 2, funcional: true};
assert("Diccionarios literales {} y acceso", persona["nombre"] == "PELE" && persona["version"] == 2 && persona["funcional"]);

doble = \x -> x * 2;
assert("Funciones anonimas (Lambdas)", doble(5) == 10 && doble(21) == 42);

res_pipe = 5 |> doble;
assert("Operador Pipeline (|>) con Lambdas", res_pipe == 10);

funcion clasificar(x) {
    si (x > 0) { retornar "pos"; }
    sino si (x < 0) { retornar "neg"; }
    sino { retornar "cero"; }
}
assert("Estructura Condicional Else-If", clasificar(5) == "pos" && clasificar(-3) == "neg" && clasificar(0) == "cero");

lista = [1, 2, 3, 4, 5];
assert("Dot Notation en Listas (.len, .head, .contains)", lista.len() == 5 && lista.head() == 1 && lista.contains(3));

assert("Primitivas Funcionales (head, tail, range)", head([1, 2]) == 1 && tail([1, 2])[0] == 2 && longitud(range(5)) == 5);

cadena_for = "";
for (c in "PELE") {
    cadena_for = cadena_for + c;
}
assert("Ciclo For-In sobre strings", cadena_for == "PELE");

va = [1, 2, 3];
vb = [4, 5, 6];
assert("Aritmetica vectorial nativa", (va + vb)[0] == 5 && (vb - va)[2] == 3 && (2 * va)[1] == 4);

funcion factorial(n) {
    si (n <= 1) { retornar 1; }
    retornar n * factorial(n - 1);
}
assert("Recursividad (Factorial)", factorial(5) == 120);
