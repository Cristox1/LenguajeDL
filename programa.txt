
mostrar(" Clasificador de medicamentos");

// 1. Cargar el dataset CSV

mostrar("Cargando archivo 'drug200.csv'");
csv_contenido = leer_archivo("drug200.csv");
df = pd_read_csv(csv_contenido);
mostrar(" => Dataset cargado con exito. Total filas: " + a_texto(longitud(df["data"])));



// 2. Preprocesamiento y Codificación One-Hot
mostrar(" Preprocesando variables categoricas One Hot");
df_proc = pd_one_hot_encode(df, "Sex");
df_proc = pd_one_hot_encode(df_proc, "BP");
df_proc = pd_one_hot_encode(df_proc, "Cholesterol");
df_oh_target = pd_one_hot_encode(df_proc, "Drug");

// Separar caracteristicas X (las primeras 9 columnas) y etiquetas y (las ultimas 5 columnas de droga)
df_X = pd_select_columns(df_oh_target, ["Age", "Sex_F", "Sex_M", "BP_HIGH", "BP_LOW", "BP_NORMAL", "Cholesterol_HIGH", "Cholesterol_NORMAL", "Na_to_K"]);
df_y = pd_select_columns(df_oh_target, ["Drug_DrugY", "Drug_drugC", "Drug_drugX", "Drug_drugA", "Drug_drugB"]);

X_raw = pd_to_numpy(df_X);
y_raw = pd_to_numpy(df_y);

// Normalizar variables continuas de entrada (Age, Na_to_K y los one-hots) mediante Min-Max
norm_res = normalizar(X_raw);
X_norm = arr_get(norm_res, 0);
mins = arr_get(norm_res, 1);
maxs = arr_get(norm_res, 2);



// 3. Inicializar y Entrenar el Perceptrón Multicapa
mostrar(" Inicializando Perceptron Multicapa");
mostrar("  -> Arquitectura: [9 entradas, 16 ocultas (ReLU), 5 salidas (MSE)]");
random_seed(42);
modelo_mlp = nn_mlp_init([9, 16, 5]);

epocas = 100;
lr = 0.05;
mostrar("  -> Comenzando entrenamiento (" + a_texto(epocas) + " epocas, lr = " + a_texto(lr) + ")...");

historia = nn_mlp_train(modelo_mlp, X_norm, y_raw, lr, epocas);

mostrar("  -> Loss Inicial (Epoca 1):   " + a_texto(arr_get(historia, 0)));
mostrar("  -> Loss Intermedio (Epoca 100): " + a_texto(arr_get(historia, 99)));
mostrar("  -> Loss Intermedio (Epoca 200): " + a_texto(arr_get(historia, 199)));
mostrar("  -> Loss Final (Epoca " + a_texto(epocas) + "):     " + a_texto(arr_get(historia, epocas - 1)));




// 4. Calcular Precision de Entrenamiento Final
mostrar("Calculando exactitud de entrenamiento ");
preds = nn_mlp_predict(modelo_mlp, X_norm);
aciertos = 0;
total_preds = longitud(preds);

por (i = 0; i < total_preds; i = i + 1) {
    // Obtener argmax de la prediccion (clasificacion)
    pred_i = arr_get(preds, i);
    max_val = arr_get(pred_i, 0);
    max_idx = 0;
    por (j = 1; j < 5; j = j + 1) {
        si (arr_get(pred_i, j) > max_val) {
            max_val = arr_get(pred_i, j);
            max_idx = j;
        }
    }
    // Obtener argmax real de target
    real_i = arr_get(y_raw, i);
    real_idx = 0;
    por (j = 0; j < 5; j = j + 1) {
        si (arr_get(real_i, j) == 1) {
            real_idx = j;
        }
    }
    si (max_idx == real_idx) {
        aciertos = aciertos + 1;
    }
}
precision = aciertos / total_preds;
mostrar("  -> Precision de clasificacion final: " + a_texto(precision * 100) + "%");

// Guardar grafico SVG de la curva de aprendizaje
plot_learning_curve(historia, "curva_aprendizaje_drogas.svg");
mostrar("  -> [SVG] Grafica guardada en 'curva_aprendizaje_drogas.svg'");





// 5. Predicciones sobre Pacientes Nuevos
mostrar(" Demostrando capacidad predictora con nuevos pacientes");

funcion crear_vector_paciente(age, sex, bp, cholesterol, na_to_k) {
    v = crear_pila();
    pila_push(v, age);
    
    si (sex == "F") { pila_push(v, 1.0); pila_push(v, 0.0); }
    sino { pila_push(v, 0.0); pila_push(v, 1.0); }
    
    si (bp == "HIGH") { pila_push(v, 1.0); pila_push(v, 0.0); pila_push(v, 0.0); }
    sino si (bp == "LOW") { pila_push(v, 0.0); pila_push(v, 1.0); pila_push(v, 0.0); }
    sino { pila_push(v, 0.0); pila_push(v, 0.0); pila_push(v, 1.0); }
    
    si (cholesterol == "HIGH") { pila_push(v, 1.0); pila_push(v, 0.0); }
    sino { pila_push(v, 0.0); pila_push(v, 1.0); }
    
    pila_push(v, na_to_k);
    retornar v;
}

funcion normalizar_punto(x, mins, maxs) {
    x_norm = crear_pila();
    por (j = 0; j < longitud(x); j = j + 1) {
        rango = arr_get(maxs, j) - arr_get(mins, j);
        si (rango == 0) { rango = 1; }
        pila_push(x_norm, (arr_get(x, j) - arr_get(mins, j)) / rango);
    }
    retornar x_norm;
}

funcion predecir_droga_paciente(modelo, age, sex, bp, cholesterol, na_to_k, mins, maxs) {
    v = crear_vector_paciente(age, sex, bp, cholesterol, na_to_k);
    v_norm = normalizar_punto(v, mins, maxs);
    
    // Realizar paso forward
    res_lote = nn_mlp_predict(modelo, [v_norm]);
    pred_vec = arr_get(res_lote, 0);
    
    // Obtener la clase predicha con mayor probabilidad (argmax)
    max_val = arr_get(pred_vec, 0);
    max_idx = 0;
    por (j = 1; j < 5; j = j + 1) {
        si (arr_get(pred_vec, j) > max_val) {
            max_val = arr_get(pred_vec, j);
            max_idx = j;
        }
    }
    
    drogas = ["DrugY", "drugC", "drugX", "drugA", "drugB"];
    retornar arr_get(drogas, max_idx);
}
